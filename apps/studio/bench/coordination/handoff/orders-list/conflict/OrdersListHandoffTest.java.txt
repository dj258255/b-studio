package com.example.api;

import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.test.web.servlet.MockMvc;

// 데이터베이스 없이 뜨도록 DB 관련 자동 설정을 끈다. 컨트롤러와 서비스 빈은 모두 불러온다.
@SpringBootTest(properties = {
        "spring.autoconfigure.exclude="
                + "org.springframework.boot.jdbc.autoconfigure.DataSourceAutoConfiguration,"
                + "org.springframework.boot.hibernate.autoconfigure.HibernateJpaAutoConfiguration,"
                + "org.springframework.boot.data.jpa.autoconfigure.JpaRepositoriesAutoConfiguration,"
                + "org.springframework.boot.flyway.autoconfigure.FlywayAutoConfiguration",
        "spring.flyway.enabled=false"
})
@AutoConfigureMockMvc
class OrdersListHandoffTest {

    @Autowired
    private MockMvc mockMvc;

    @Test
    void returnsThreeSampleOrders() throws Exception {
        mockMvc.perform(get("/api/orders"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.length()").value(3))
                .andExpect(jsonPath("$[0].customerName").value("김민수"))
                .andExpect(jsonPath("$[1].customerName").value("이영희"))
                .andExpect(jsonPath("$[2].customerName").value("박철호"))
                .andExpect(jsonPath("$[0].id").exists())
                .andExpect(jsonPath("$[1].id").exists())
                .andExpect(jsonPath("$[2].id").exists())
                .andExpect(jsonPath("$[0].amount").exists())
                .andExpect(jsonPath("$[1].amount").exists())
                .andExpect(jsonPath("$[2].amount").exists())
                .andExpect(jsonPath("$[0].status").exists())
                .andExpect(jsonPath("$[1].status").exists())
                .andExpect(jsonPath("$[2].status").exists());
    }
}
